Fórmulas Matemáticas del Modelo M/M/1/K Implementadas en el Código

A continuación se expone la estructura algorítmica que ejecuta los cálculos del sistema en tiempo real: 

// ============================================================================
// 1. ECUACIÓN DEL TIEMPO DE SERVICIO T_s(Q_i, m) SEGÚN BLOQUES Y CARGUEROS
// Restricción operativa: Q_i mínimo de 1.000 bloques para evitar tiempos negativos.
// Pendiente medida en campo (3 cargueros): m = (110 - 50) / (3000 - 1500) = 0.04 min/bloque
// Recta base (3 cargueros): T_s(Q_i) = 0.04 * Q_i - 10 minutos
// Escalamiento proporcional para m cargueros: T_s(Q_i, m) = (3 / m) * (0.04 * Q_i - 10)
// ============================================================================
function calcularTiempoPorPedido(bloques, cargueros) {
    const bloquesValidados = Math.max(1000, bloques); // Límite mínimo: 1.000 bloques
    const tiempoTresCargueros = (0.04 * bloquesValidados) - 10;
    return Math.max(1, Math.round((3 / cargueros) * tiempoTresCargueros));
}

// ============================================================================
// 2. MOTOR MATEMÁTICO DE ESTADO ESTABLE PARA EL MODELO M/M/1/K
// Calcula: lambda (Landa), mu (Niu), 1/lambda, 1/mu, rho, P0, Pn, PK, lambda_ef, L, Lq, W, Wq
// ============================================================================
function calcularVariablesMM1K(totalLlegadas, horasEfectivas, tiempoServicioMin, K) {
    // Tasa media de llegada Landa (camiones/hora) y tiempo entre llegadas (1 / Landa)
    const lambda = totalLlegadas / horasEfectivas;
    const tiempoEntreLlegadasHoras = 1 / lambda;
    const tiempoEntreLlegadasMin = 60 / lambda;

    // Tasa media de servicio Niu (camiones/hora) -> conversión con 60 min = 1 hora
    const mu = 60 / tiempoServicioMin;
    const tiempoServicioHoras = 1 / mu;

    // Factor de utilización o intensidad de tráfico Rho (ρ = λ / μ)
    const rho = lambda / mu;

    // Probabilidad de que el sistema esté vacío (P_0)
    let P0;
    if (Math.abs(rho - 1.0) < 1e-9) {
        P0 = 1 / (K + 1);
    } else {
        P0 = (1 - rho) / (1 - Math.pow(rho, K + 1));
    }

    // Distribución de probabilidad P_n = P_0 * (ρ^n) para cada estado n = 0, 1, ..., K
    const Pn = [];
    for (let n = 0; n <= K; n++) {
        Pn.push(P0 * Math.pow(rho, n));
    }

    // Probabilidad de bloqueo / % de pérdida de clientes en el estado límite (P_K)
    const PK = Pn[K];

    // Tasa efectiva de llegada (Landa efectiva) y camiones despachados en la jornada
    const lambdaEfectiva = lambda * (1 - PK);
    const camionesDespachadosJornada = lambdaEfectiva * horasEfectivas;

    // Número promedio de camiones en el sistema (L) y esperando en la cola (L_q)
    let L;
    if (Math.abs(rho - 1.0) < 1e-9) {
        L = K / 2;
    } else {
        L = (rho / (1 - rho)) - (((K + 1) * Math.pow(rho, K + 1)) / (1 - Math.pow(rho, K + 1)));
    }
    const Lq = L - (1 - P0);

    // Tiempo promedio en el sistema (W) y en la fila (W_q) por las relaciones de Little
    const W_horas = L / lambdaEfectiva;
    const Wq_horas = Lq / lambdaEfectiva;

    return {
        lambda, mu, tiempoEntreLlegadasHoras, tiempoEntreLlegadasMin,
        tiempoServicioHoras, tiempoServicioMin, rho, P0, Pn, PK,
        lambdaEfectiva, camionesDespachadosJornada, L, Lq, W_horas, Wq_horas
    };
}
