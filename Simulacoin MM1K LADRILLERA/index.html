Sustentación del modelo M/M/1/K programado en JavaScript ES6 — Archivo: js/math/mm1kEngine.js

1. Tiempo de Servicio — calcularTiempoPorPedido(bloques, cargueros)
// ============================================================================ // ECUACIÓN DEL TIEMPO DE CARGUE (SERVICIO) POR PEDIDO // ============================================================================ // Ecuación base obtenida por medición directa en campo con m = 3 cargueros: // T_s(Q_i) = 0.04 × Q_i − 10 (en minutos) // // Escalamiento por número de cargueros (m): // T_s(Q_i, m) = (3 / m) × (0.04 × Q_i − 10) (en minutos) // // Ejemplo con m = 5: // T_s = (3/5) × (0.04 × Q_i − 10) = 0.024 × Q_i − 6 // // VALIDACIÓN: Q_i mínimo = 1000 bloques (evita tiempos negativos) // ============================================================================ export function calcularTiempoPorPedido(bloques, cargueros) { const Q = Math.max(1000, bloques); // Validación: mín 1000 bloques const m = Math.max(1, cargueros); // Validación: mín 1 carguero const Ts = (3 / m) * (0.04 * Q - 10); // Ecuación escalada por m return Math.max(0, Ts); // Nunca negativo }
2. Variables de Estado Estable — calcularVariablesMM1K({...})
// ============================================================================ // MOTOR MATEMÁTICO DEL MODELO DE COLAS M/M/1/K // ============================================================================ // Parámetros de entrada: // totalLlegadas → Total de camiones que llegan en el día // horasEfectivas → Jornada laboral efectiva H_ef (horas) // tiempoServicioMin → Tiempo medio de servicio T_s (minutos) // capacidadK → Capacidad máxima del sistema K // ============================================================================ export function calcularVariablesMM1K({ totalLlegadas, horasEfectivas, tiempoServicioMin, capacidadK }) { // 1. Tasa media de llegada λ = Total_Llegadas / H_ef const lambda = totalLlegadas / horasEfectivas; // cam/hora // 2. Tiempo medio entre llegadas const T_a_h = 1 / lambda; // horas const T_a_min = T_a_h * 60; // minutos // 3. Tasa media de servicio μ = 60 / T_s_min const mu = 60 / tiempoServicioMin; // cam/hora // 4. Tiempo medio de servicio const T_s_h = tiempoServicioMin / 60; // horas // 7. Factor de utilización ρ = λ / μ const rho = lambda / mu; // 8. Probabilidad de servidor vacío P₀ let P_0; if (Math.abs(rho - 1) < 1e-10) { P_0 = 1 / (capacidadK + 1); // Caso ρ = 1 } else { P_0 = (1 - rho) / (1 - Math.pow(rho, capacidadK + 1)); // Caso general } // 9. Distribución: P_n = P₀ × ρⁿ para n = 0, 1, ..., K const distribucionPn = []; for (let n = 0; n <= capacidadK; n++) { const P_n = P_0 * Math.pow(rho, n); distribucionPn.push({ n, P_n, porcentaje: P_n * 100 }); } // 10. Probabilidad de bloqueo P_K = P₀ × ρᴷ const P_K = distribucionPn[capacidadK].P_n; // 11. Tasa efectiva λ_ef = λ × (1 − P_K) const lambda_ef = lambda * (1 - P_K); // 12. Despacho diario = λ_ef × H_ef const despacho = lambda_ef * horasEfectivas; // 13. Camiones promedio en el sistema L let L; if (Math.abs(rho - 1) < 1e-10) { L = capacidadK / 2; } else { L = (rho / (1 - rho)) - ((capacidadK + 1) * Math.pow(rho, capacidadK + 1)) / (1 - Math.pow(rho, capacidadK + 1)); } // 14. Camiones promedio en la fila L_q = L − (1 − P₀) const L_q = L - (1 - P_0); // 15. Tiempo promedio en el sistema W = L / λ_ef const W_h = L / lambda_ef; // horas const W_min = W_h * 60; // minutos // 16. Tiempo promedio en la fila W_q = L_q / λ_ef const W_q_h = L_q / lambda_ef; // horas const W_q_min = W_q_h * 60; // minutos return { lambda, T_a_h, T_a_min, mu, T_s_h, rho, P_0, distribucionPn, P_K, lambda_ef, despacho, L, L_q, W_h, W_min, W_q_h, W_q_min, H_ef: horasEfectivas, T_s_min: tiempoServicioMin, c: 1, K: capacidadK, totalLlegadas }; }

Concepto Fórmula

ρ (Factor de utilización)λ / μ
P₀ (Servidor vacío)(1 − ρ) / (1 − ρ^(K+1)) si ρ ≠ 1 ‖ 1/(K+1) si ρ = 1
P_n (Distribución)P₀ × ρⁿ para n = 0, 1, ..., K
P_K (Bloqueo)P₀ × ρᴷ
λ_ef (Tasa efectiva)λ × (1 − P_K)
